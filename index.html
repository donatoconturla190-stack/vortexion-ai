Integración API Enterprise

v1 · Internal

Endpoint compatible con la especificación OpenAI Chat Completions. Autenticación mediante Bearer token. Uso restringido a entornos autorizados.

import requests

API_KEY = "sk-TU_CLAVE_AQUI"
BASE_URL = "http://vortexion.labsia.internal/v1"   # o https:// si tiene certificado

headers = {
    "Authorization": f"Bearer {API_KEY}",
    "Content-Type": "application/json"
}

response = requests.post(
    f"{BASE_URL}/chat/completions",
    headers=headers,
    json={
        "model": "gpt-4o-mini",
        "messages": [{"role": "user", "content": "Hola"}]
    }
)

print(response.status_code)
print(response.json())
